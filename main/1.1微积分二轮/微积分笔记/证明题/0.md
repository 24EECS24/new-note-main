证明不等式：$\ln(1+e^x)-x>\dfrac{1}{e^x+1}$，$x\in(-\infty,+\infty)$。

$$
\begin{align*}
证明：& \\[4pt]
&\ln(1+e^x) - \ln e^x > \frac{1}{e^x + 1}  \\[4pt]
&\ln(1+e^{-x}) > \frac{1}{e^x + 1}  \\[4pt]
&设e^x=t， \\[4pt]
&\ln\left(1+\frac{1}{t}\right) > \frac{1}{t+1}  \\[4pt]
&g(t) = \ln\left(1+\frac{1}{t}\right) - \frac{1}{t+1}  \\[4pt]
&\begin{aligned}
g'(t) &= \frac{1}{1+\frac{1}{t}} \cdot \left(-\frac{1}{t^2}\right) + \frac{1}{(t+1)^2} \\[6pt]
&= \frac{-\frac{1}{t^2}}{1+\frac{1}{t}} + \frac{1}{(t+1)^2} \\[6pt]
&= \frac{-\frac{1}{t}}{t+1} + \frac{1}{(t+1)^2} \\[6pt]
&= \frac{-\left(1+\frac{1}{t}\right) + 1}{(t+1)^2}
\end{aligned} \\[4pt]

&\because \ -\left(1+\frac{1}{t}\right)+1 = -\frac{1}{t} < 0，(t+1)^2 > 0， \\[4pt]
&\therefore \ g'(t) < 0 \\[4pt]
&\because \ \lim_{t \to +\infty} g(t) = 0 \\[4pt]
&\therefore \ x\in(-\infty,+\infty)，\ln(1+e^x) - x > \frac{1}{e^x + 1}   \\[4pt]

\end{align*}
$$
