\begin{document}

\begin{tikzpicture}[x=1cm,y=1cm,scale=1.0]
    % axes
    \draw[->] (-0.3,0) -- (5,0) node[right,font=\small] {$x$};
    \draw[->] (0,-0.3) -- (0,3) node[above,font=\small] {$y$};
    \node[below left,font=\small] at (0,0) {$O$};

    % precomputed: f(1)=1, f(4)=2, f(1.5)=1.198, chord slope=1/3
    \draw[dashed,gray] (1,0) -- (1,1);
    \draw[dashed,gray] (4,0) -- (4,2);
    \draw[dashed,gray] (1.5,0) -- (1.5,1.198);
    \draw (1,0) node[below,font=\small] {$a$};
    \draw (4,0) node[below,font=\small] {$b$};
    \draw (1.5,0) node[below,font=\small] {$\xi$};

    % chord
    \draw[chordred,thick] (1,1) -- (4,2) node[above,pos=0.8,font=\small,sloped] {chord};

    % curve
    \draw[curveblue,thick,smooth,samples=100,domain=1:4]
        plot(\x, {0.05*(\x-1)*(\x-2)*(\x-4) + (1/3)*\x + 2/3});

    % tangent at xi
    \draw[tangentgreen,thick] (0.7,0.931) node[above left,font=\small,sloped] {tangent} -- (2.7,1.598);

    % points
    \fill[curveblue] (1,1) circle (1.2pt) node[below left,font=\small] {$(a,f(a))$};
    \fill[curveblue] (4,2) circle (1.2pt) node[above right,font=\small] {$(b,f(b))$};
    \fill[tangentgreen] (1.5,1.198) circle (1.2pt);

    \node[font=\small] at (2.5,2.7) {$f'(\xi)=\frac{f(b)-f(a)}{b-a}$};
\end{tikzpicture}
\end{document}

### 三、补充优化图（贴合例6.11的柯西中值应用）
针对例题中$g(x)=\ln x$的场景，绘制参数方程$x=\ln t, y=f(t)$（取$f(t)=t$即曲线$y=e^x$）的示意图，更直观展示例题结论：
```tikz
\begin{document}
\definecolor{curveblue}{rgb}{0.2,0.3,0.8}
\definecolor{chordred}{rgb}{0.8,0.2,0.2}
\definecolor{tangentgreen}{rgb}{0,0.6,0}
\begin{tikzpicture}[x=2cm,y=0.5cm,scale=1.0]
    % 坐标轴：x=ln t ∈[0,2]（对应t∈[1,e²]），y=f(t)=t
    \draw[->] (-0.3,0) -- (2.5,0) node[right,font=\small] {$x=\ln t$};
    \draw[->] (0,-0.5) -- (0,8) node[above,font=\small] {$y=f(t)=t$};
    \node[below left,font=\small] at (0,0) {$O$};

    % 例题参数：a=1, b=e², ξ=(e²-1)/2≈3.194
    \def\ta{1}
    \def\tb{7.389}
    \def\txi{3.194}
    \def\ga{ln(\ta)} % ln1=0
    \def\gb{ln(\tb)} % lne²=2
    \def\gxi{ln(\txi)} % lnξ≈1.162
    \def\fa{\ta} % f(a)=1
    \def\fb{\tb} % f(b)=e²≈7.389
    \def\fxi{\txi} % f(ξ)=ξ

    % 辅助线与刻度
    \draw (\ga,0) node[below left,font=\small] {$\ln a=0$};
    \draw (\gb,0) node[below,font=\small] {$\ln b=2$};
    \draw (\gxi,0) node[below,font=\small] {$\ln\xi$};
    \draw (0,\fa) node[left,font=\small] {$f(a)=1$};
    \draw (0,\fb) node[left,font=\small] {$f(b)=e^2$};
    \draw[dashed,gray] (\ga,0) -- (\ga,\fa);
    \draw[dashed,gray] (\gb,0) -- (\gb,\fb);
    \draw[dashed,gray] (\gxi,0) -- (\gxi,\fxi);
    \draw[dashed,gray] (0,\fxi) -- (\gxi,\fxi);

    % 弦：连接(ln a,f(a))与(ln b,f(b))
    \def\slope{(\fb-\fa)/(\gb-\ga)} % =ξ，符合例题结论
    \draw[chordred,thick] (\ga,\fa) -- (\gb,\fb)
        node[above right,pos=0.3,font=\small,sloped] {chord};

    % 曲线y=e^x（对应参数方程x=ln t,y=t）
    \draw[curveblue,thick,smooth,samples=100,domain=0:2] 
        plot(\x, {exp(\x)});
    \node[curveblue,above right,font=\small] at (1.5,{exp(1.5)}) {$y=e^x$};

    % ξ点处的切线
    \draw[tangentgreen,thick] ({\gxi-0.3}, {\fxi - 0.3*\slope}) -- ({\gxi+0.5}, {\fxi + 0.5*\slope})
        node[above right,font=\small,sloped] {tangent};

    % 点标记
    \fill[curveblue] (\ga,\fa) circle (1.2pt);
    \fill[curveblue] (\gb,\fb) circle (1.2pt);
    \fill[tangentgreen] (\gxi,\fxi) circle (1.2pt) node[above right,font=\small] {$(\ln\xi,\xi)$};

    % 例题结论公式
    \node[font=\small,align=left] at (1.2,-1.5) 
        {$\frac{f(b)-f(a)}{\ln b-\ln a}=\frac{f'(\xi)}{(\ln\xi)'}=\xi f'(\xi)$\\
        $f(b)-f(a)=\xi f'(\xi)\ln\frac{b}{a}$};
\end{tikzpicture}
\end{document}
```
