### 二、笔记原图的TikZJax实现
#### 1. 拉格朗日中值定理几何示意图
```tikz
\begin{document}
\definecolor{curveblue}{rgb}{0.2,0.3,0.8}
\definecolor{chordred}{rgb}{0.8,0.2,0.2}
\definecolor{tangentgreen}{rgb}{0,0.6,0}
\begin{tikzpicture}[x=1cm,y=1cm,scale=1.0]
    % 坐标轴
    \draw[->] (-0.3,0) -- (5,0) node[right,font=\small] {$x$};
    \draw[->] (0,-0.3) -- (0,3) node[above,font=\small] {$y$};
    \node[below left,font=\small] at (0,0) {$O$};

    % 坐标点定义（纯数值宏，无数字/希腊字母冲突）
    \def\pa{1}    % a点x坐标
    \def\pb{4}    % b点x坐标
    \def\px{1.5}  % ξ点x坐标
    \def\fa{0.05*(\pa-1)*(\pa-2)*(\pa-4) + (1/3)*\pa + 2/3} % f(a)
    \def\fb{0.05*(\pb-1)*(\pb-2)*(\pb-4) + (1/3)*\pb + 2/3} % f(b)
    \def\fy{0.05*(\px-1)*(\px-2)*(\px-4) + (1/3)*\px + 2/3} % f(ξ)

    % 辅助虚线与刻度
    \draw[dashed,gray] (\pa,0) -- (\pa,\fa);
    \draw[dashed,gray] (\pb,0) -- (\pb,\fb);
    \draw[dashed,gray] (\px,0) -- (\px,\fy);
    \draw (\pa,0) node[below,font=\small] {$a$};
    \draw (\pb,0) node[below,font=\small] {$b$};
    \draw (\px,0) node[below,font=\small] {$\xi$};

    % 连接端点的弦
    \def\slope{(\fb-\fa)/(\pb-\pa)}
    \draw[chordred,thick] (\pa,\fa) -- (\pb,\fb) node[above right,pos=0.5,font=\small,sloped] {chord};

    % 函数曲线
    \draw[curveblue,thick,smooth,samples=100,domain=\pa:\pb] 
        plot(\x, {0.05*(\x-1)*(\x-2)*(\x-4) + (1/3)*\x + 2/3});

    % ξ点处的平行切线
    \draw[tangentgreen,thick] ({\px-0.8}, {\fy - 0.8*\slope}) -- ({\px+1.2}, {\fy + 1.2*\slope})
        node[above right,font=\small,sloped] {tangent};

    % 点标记
    \fill[curveblue] (\pa,\fa) circle (1.2pt) node[below left,font=\small] {$(a,f(a))$};
    \fill[curveblue] (\pb,\fb) circle (1.2pt) node[above right,font=\small] {$(b,f(b))$};
    \fill[tangentgreen] (\px,\fy) circle (1.2pt);

    % 中值定理公式
    \node[font=\small] at (2.5,2.7) {$f'(\xi)=\frac{f(b)-f(a)}{b-a}$};
\end{tikzpicture}
\end{document}
```

#### 2. 柯西中值定理（参数方程形式）示意图
```tikz
\begin{document}
\definecolor{curveblue}{rgb}{0.2,0.3,0.8}
\definecolor{chordred}{rgb}{0.8,0.2,0.2}
\definecolor{tangentgreen}{rgb}{0,0.6,0}
\begin{tikzpicture}[x=1cm,y=1cm,scale=1.0]
    % 坐标轴（参数方程坐标：x=g(t), y=f(t)）
    \draw[->] (-0.3,0) -- (5,0) node[right,font=\small] {$g(t)$};
    \draw[->] (0,-0.3) -- (0,3) node[above,font=\small] {$f(t)$};
    \node[below left,font=\small] at (0,0) {$O$};

    % 参数点定义
    \def\ta{1}    % 参数t=a
    \def\tb{4}    % 参数t=b
    \def\tx{1.5}  % 参数t=ξ
    \def\ga{\ta}  % g(a)（示意取g(t)=t，对应拉格朗日为柯西特例）
    \def\gb{\tb}  % g(b)
    \def\gx{\tx}  % g(ξ)
    \def\fa{0.05*(\ta-1)*(\ta-2)*(\ta-4) + (1/3)*\ta + 2/3} % f(a)
    \def\fb{0.05*(\tb-1)*(\tb-2)*(\tb-4) + (1/3)*\tb + 2/3} % f(b)
    \def\fx{0.05*(\tx-1)*(\tx-2)*(\tx-4) + (1/3)*\tx + 2/3} % f(ξ)

    % 辅助虚线与刻度
    \draw[dashed,gray] (\ga,0) -- (\ga,\fa);
    \draw[dashed,gray] (\gb,0) -- (\gb,\fb);
    \draw[dashed,gray] (\gx,0) -- (\gx,\fx);
    \draw (\ga,0) node[below,font=\small] {$a$};
    \draw (\gb,0) node[below,font=\small] {$b$};
    \draw (\gx,0) node[below,font=\small] {$\xi$};

    % 参数曲线的弦
    \def\slope{(\fb-\fa)/(\gb-\ga)}
    \draw[chordred,thick] (\ga,\fa) -- (\gb,\fb)
        node[above right,pos=0.5,font=\small,sloped] {$k$};

    % 参数方程曲线
    \draw[curveblue,thick,smooth,samples=100,domain=\ta:\tb] 
        plot(\x, {0.05*(\x-1)*(\x-2)*(\x-4) + (1/3)*\x + 2/3});

    % 参数曲线在t=ξ处的切线
    \draw[tangentgreen,thick] ({\gx-0.8}, {\fx - 0.8*\slope}) -- ({\gx+1.2}, {\fx + 1.2*\slope})
        node[above right,font=\small,sloped] {$\frac{dy}{dx}$};

    % 点标记
    \fill[curveblue] (\ga,\fa) circle (1.2pt) node[below left,font=\small] {$(g(a),f(a))$};
    \fill[curveblue] (\gb,\fb) circle (1.2pt) node[above right,font=\small] {$(g(b),f(b))$};
    \fill[tangentgreen] (\gx,\fx) circle (1.2pt);

    % 参数方程与柯西中值公式
    \node[font=\small,right] at (3.5,2.7) {$\left\{\begin{array}{l}x=g(t)\\y=f(t)\end{array}\right.$};
    \node[font=\small] at (2.5,-0.8) {$k=\frac{f(b)-f(a)}{g(b)-g(a)}=\left.\frac{dy}{dx}\right|_{\xi}=\frac{f'(\xi)}{g'(\xi)}$};
\end{tikzpicture}
\end{document}
```

### 三、补充优化图（贴合例6.11的柯西中值应用）
针对例题中$g(x)=\ln x$的场景，绘制参数方程$x=\ln t, y=f(t)$（取$f(t)=t$即曲线$y=e^x$）的示意图，更直观展示例题结论：
```tikz
\begin{document}
\definecolor{curveblue}{rgb}{0.2,0.3,0.8}
\definecolor{chordred}{rgb}{0.8,0.2,0.2}
\definecolor{tangentgreen}{rgb}{0,0.6,0}
\begin{tikzpicture}[x=2cm,y=0.5cm,scale=1.0]
    % 坐标轴：x=ln t ∈[0,2]（对应t∈[1,e²]），y=f(t)=t
    \draw[->] (-0.3,0) -- (2.5,0) node[right,font=\small] {$x=\ln t$};
    \draw[->] (0,-0.5) -- (0,8) node[above,font=\small] {$y=f(t)=t$};
    \node[below left,font=\small] at (0,0) {$O$};

    % 例题参数：a=1, b=e², ξ=(e²-1)/2≈3.194
    \def\ta{1}
    \def\tb{7.389}
    \def\txi{3.194}
    \def\ga{ln(\ta)} % ln1=0
    \def\gb{ln(\tb)} % lne²=2
    \def\gxi{ln(\txi)} % lnξ≈1.162
    \def\fa{\ta} % f(a)=1
    \def\fb{\tb} % f(b)=e²≈7.389
    \def\fxi{\txi} % f(ξ)=ξ

    % 辅助线与刻度
    \draw (\ga,0) node[below left,font=\small] {$\ln a=0$};
    \draw (\gb,0) node[below,font=\small] {$\ln b=2$};
    \draw (\gxi,0) node[below,font=\small] {$\ln\xi$};
    \draw (0,\fa) node[left,font=\small] {$f(a)=1$};
    \draw (0,\fb) node[left,font=\small] {$f(b)=e^2$};
    \draw[dashed,gray] (\ga,0) -- (\ga,\fa);
    \draw[dashed,gray] (\gb,0) -- (\gb,\fb);
    \draw[dashed,gray] (\gxi,0) -- (\gxi,\fxi);
    \draw[dashed,gray] (0,\fxi) -- (\gxi,\fxi);

    % 弦：连接(ln a,f(a))与(ln b,f(b))
    \def\slope{(\fb-\fa)/(\gb-\ga)} % =ξ，符合例题结论
    \draw[chordred,thick] (\ga,\fa) -- (\gb,\fb)
        node[above right,pos=0.3,font=\small,sloped] {chord};

    % 曲线y=e^x（对应参数方程x=ln t,y=t）
    \draw[curveblue,thick,smooth,samples=100,domain=0:2] 
        plot(\x, {exp(\x)});
    \node[curveblue,above right,font=\small] at (1.5,{exp(1.5)}) {$y=e^x$};

    % ξ点处的切线
    \draw[tangentgreen,thick] ({\gxi-0.3}, {\fxi - 0.3*\slope}) -- ({\gxi+0.5}, {\fxi + 0.5*\slope})
        node[above right,font=\small,sloped] {tangent};

    % 点标记
    \fill[curveblue] (\ga,\fa) circle (1.2pt);
    \fill[curveblue] (\gb,\fb) circle (1.2pt);
    \fill[tangentgreen] (\gxi,\fxi) circle (1.2pt) node[above right,font=\small] {$(\ln\xi,\xi)$};

    % 例题结论公式
    \node[font=\small,align=left] at (1.2,-1.5) 
        {$\frac{f(b)-f(a)}{\ln b-\ln a}=\frac{f'(\xi)}{(\ln\xi)'}=\xi f'(\xi)$\\
        $f(b)-f(a)=\xi f'(\xi)\ln\frac{b}{a}$};
\end{tikzpicture}
\end{document}
```
