### 笔记原图的TikZ绘制
```tikz
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm,>=stealth,scale=1.1]
% 拉格朗日中值定理示意图
\begin{scope}
    % 坐标轴
    \draw[->] (-0.3,0) -- (5,0) node[right,font=\small] {$x$};
    \draw[->] (0,-0.3) -- (0,3) node[above,font=\small] {$y$};
    \node[below left,font=\small] at (0,0) {$O$};
    % x轴刻度
    \draw (1,-0.08) -- (1,0.08) node[below,font=\small] {$a$};
    \def\xi{1.6}
    \draw (\xi,-0.08) -- (\xi,0.08) node[below,font=\small] {$\xi$};
    \draw (4,-0.08) -- (4,0.08) node[below,font=\small] {$b$};
    % 端点
    \fill (1,1) circle (1pt) node[above left,font=\small] {$(a,f(a))$};
    \fill (4,2) circle (1pt) node[above right,font=\small] {$(b,f(b))$};
    % 弦AB
    \draw[thick] (1,1) -- (4,2) node[midway,above,sloped,font=\small] {$\frac{f(b)-f(a)}{b-a}$};
    % 曲线f(x)
    \draw[thick,smooth,samples=100,domain=1:4] 
        plot(\x, {0.1*(\x-1)*(\x-2.5)*(\x-4) + 1 + (1/3)*(\x-1)});
    % 切点ξ
    \def\fxi{0.1*(\xi-1)*(\xi-2.5)*(\xi-4) + 1 + (1/3)*(\xi-1)}
    \fill (\xi,\fxi) circle (1pt);
    % 切线
    \draw[thick,dashed] ({\xi-0.6}, {\fxi - (1/3)*0.6}) -- ({\xi+0.8}, {\fxi + (1/3)*0.8})
        node[above right,font=\small] {$f'(\xi)$};
    % 标注
    \node[font=\small] at (3.5,2.7) {Lagrange MVT};
\end{scope}

% 柯西中值定理（参数方程形式）示意图
\begin{scope}[shift={(0,-5)}]
    % 坐标轴
    \draw[->] (-0.3,0) -- (5,0) node[right,font=\small] {$g(t)$};
    \draw[->] (0,-0.3) -- (0,3) node[above,font=\small] {$f(t)$};
    \node[below left,font=\small] at (0,0) {$O$};
    % 横轴刻度（参数t的取值）
    \draw (1,-0.08) -- (1,0.08) node[below,font=\small] {$a$};
    \def\xi{1.6}
    \draw (\xi,-0.08) -- (\xi,0.08) node[below,font=\small] {$\xi$};
    \draw (4,-0.08) -- (4,0.08) node[below,font=\small] {$b$};
    % 端点
    \fill (1,1) circle (1pt) node[above left,font=\small] {$(g(a),f(a))$};
    \fill (4,2) circle (1pt) node[above right,font=\small] {$(g(b),f(b))$};
    % 弦
    \draw[thick] (1,1) -- (4,2) node[midway,above,sloped,font=\small] {$k=\frac{f(b)-f(a)}{g(b)-g(a)}$};
    % 参数曲线
    \draw[thick,smooth,samples=100,domain=1:4] 
        plot(\x, {0.1*(\x-1)*(\x-2.5)*(\x-4) + 1 + (1/3)*(\x-1)});
    % 切点ξ
    \def\fxi{0.1*(\xi-1)*(\xi-2.5)*(\xi-4) + 1 + (1/3)*(\xi-1)}
    \fill (\xi,\fxi) circle (1pt);
    % 切线
    \draw[thick,dashed] ({\xi-0.6}, {\fxi - (1/3)*0.6}) -- ({\xi+0.8}, {\fxi + (1/3)*0.8})
        node[above right,font=\small] {$\frac{dy}{dx}\bigg|_{\xi}=\frac{f'(\xi)}{g'(\xi)}$};
    % 参数方程标注
    \node[font=\small,right] at (4.2,1.5) {$\begin{cases}x=g(t)\\y=f(t)\end{cases}$};
    % 标注
    \node[font=\small] at (3.5,2.7) {Cauchy MVT};
\end{scope}
\end{tikzpicture}
\end{document}
```

### 补充辅助理解图（例6.11的辅助函数$g(x)=\ln x$）
该图展示证明中选取的辅助对数函数，直观体现其导数$g'(x)=\frac{1}{x}$的形式，对应结论中的$\xi$因子，帮助理解证明思路：
```tikz
\begin{document}
\definecolor{curveblue}{rgb}{0.2,0.3,0.8}
\definecolor{tangentred}{rgb}{0.8,0.2,0.2}
\begin{tikzpicture}[x=1cm,y=1cm,>=stealth,scale=1.2]
    % 坐标轴
    \draw[->] (-0.3,0) -- (5,0) node[right,font=\small] {$x$};
    \draw[->] (0,-2) -- (0,2) node[above,font=\small] {$y$};
    \node[below left,font=\small] at (0,0) {$O$};
    % 区间刻度
    \def\a{1}
    \def\b{4}
    \draw (\a,-0.08) -- (\a,0.08) node[below,font=\small] {$a$};
    \draw (\b,-0.08) -- (\b,0.08) node[below,font=\small] {$b$};
    % 绘制g(x)=lnx
    \draw[thick,curveblue,smooth,samples=100,domain=0.5:4.5] 
        plot(\x, {ln(\x)}) node[above right,font=\small] {$g(x)=\ln x$};
    % 标注导数几何意义
    \def\xt{2.5}
    \draw[thick,dashed,tangentred] ({\xt-0.8}, {ln(\xt) - 0.8/\xt}) -- ({\xt+0.8}, {ln(\xt) + 0.8/\xt});
    \node[tangentred,font=\small] at (3.5,1.3) {$g'(x)=\frac{1}{x}$};
    % 端点辅助线
    \draw[dashed,gray] (\a,0) -- (\a,{ln(\a)});
    \draw[dashed,gray] (\b,0) -- (\b,{ln(\b)});
    \fill (\a,{ln(\a)}) circle (1pt) node[above left,font=\small] {$(a,\ln a)$};
    \fill (\b,{ln(\b)}) circle (1pt) node[above right,font=\small] {$(b,\ln b)$};
\end{tikzpicture}
\end{document}
```
